<script>
  // Año actual en el pie de página
  document.getElementById("anio").textContent = new Date().getFullYear();

  // Estado "Abierto / Cerrado" según el horario
  (function () {
    var ahora = new Date();
    var dia = ahora.getDay(); // 0 = domingo
    var minutos = ahora.getHours() * 60 + ahora.getMinutes();
    var abre, cierra;
    if (dia === 0) { abre = 9 * 60; cierra = 13 * 60; }
    else if (dia === 6) { abre = 8 * 60; cierra = 18 * 60; }
    else { abre = 8 * 60; cierra = 20 * 60; }
    var abierto = minutos >= abre && minutos < cierra;
    var caja = document.getElementById("estado");
    caja.classList.add(abierto ? "abierto" : "cerrado");
    document.getElementById("estado-texto").textContent = abierto ? "Abierto ahora" : "Cerrado ahora";
  })();
</script>
